function(hezarfen_sdk_target name)
  target_include_directories(${name} PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS})
  target_link_libraries(${name} PRIVATE VSI::simVis VSI::simUtil ${OPENSCENEGRAPH_LIBRARIES})
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /utf-8 /wd4251 /wd4275)
    target_compile_definitions(${name} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  endif()
  set_target_properties(${name} PROPERTIES VS_DEBUGGER_ENVIRONMENT "PATH=${SIMDIS_SDK_DIR}/bin;%PATH%\nSIMDIS_SDK_FILE_PATH=${SIMDIS_DATA_DIR}")
endfunction()
